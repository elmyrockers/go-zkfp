\section{Overview}
\texttt{go-zkfp} (\url{github.com/elmyrockers/go-zkfp}) is a Go binding for \texttt{libzkfp.dll}, the ZKFinger Reader SDK from ZKTeco. It allows Go programs to enumerate ZKTeco fingerprint readers, capture images and templates, and register, identify, and match fingerprints. This document describes the Go API, mapping one-to-one to the C functions of the ZKFinger Reader SDK (C API Version 2.0).

\section{Disclaimer}
\texttt{go-zkfp} is an independent open-source wrapper and does not include the ZKTeco SDK binaries. \texttt{libzkfp.dll} must be obtained by installing the official ZKFinger SDK from ZKTeco. Use of the SDK is subject to ZKTeco's licensing terms.

\section{System Requirements}
\begin{enumerate}
    \item Operating system: Windows XP or a later version
    \item Go 1.27.1 or later
    \item ZKFinger SDK 5.x / ZKOnline SDK 5.x installed (providing \texttt{libzkfp.dll} and reader drivers)
    \item The architecture of the Go program (\texttt{GOARCH=386} or \texttt{amd64}) must match the architecture of \texttt{libzkfp.dll}
    \item cgo and a C compiler are \textbf{not} required: the library is loaded at runtime via \texttt{golang.org/x/sys/windows} (\texttt{windows.NewLazyDLL}) utilizing standard calling conventions required by the SDK.
\end{enumerate}

\section{Installation and Deployment}
\begin{enumerate}
    \item Install ZKFinger SDK 5.x / ZKOnline SDK 5.x.
    \item Add the package to your module:
\begin{lstlisting}
go get github.com/elmyrockers/go-zkfp
\end{lstlisting}
    \item Make \texttt{libzkfp.dll} loadable by placing it next to your executable or in a directory on your system \texttt{PATH}.
    \item Import the package in your code:
\begin{lstlisting}
import "github.com/elmyrockers/go-zkfp"
\end{lstlisting}
\end{enumerate}

\section{Description of Go Interfaces}

\subsection{Design Conventions}
\begin{itemize}
    \item \textbf{Errors.} Functions return a standard Go \texttt{error} instead of raw integer return codes. Non-zero SDK results return values of type \texttt{zkfp.Error} (see Appendix 2), allowing direct comparison with \texttt{errors.Is} or \texttt{errors.As}.
    \item \textbf{Handles.} C \texttt{HANDLE} types are encapsulated into clean Go types: \texttt{*Device} (for reader instances) and \texttt{*DB} (for algorithm caches/databases).
    \item \textbf{Buffers.} C pointer-and-size parameter pairs are represented as native Go slices (\texttt{[]byte}). Output buffers are automatically allocated and trimmed to the actual data size returned by the SDK.
    \item \textbf{Cleanup.} \texttt{Close()} and \texttt{Free()} methods are implemented safely and idempotently.
\end{itemize}

\section{Global Library Functions}

\subsection{Init}
\textbf{[Function]} \texttt{func Init() error} \\
\textbf{[Purpose]} Initializes the ZKFinger SDK resources. Must be called once before invoking other methods. \\
\textbf{[Parameter Description]} None \\
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{nil}: Succeeded (including code 1 indicating already initialized)
    \item \texttt{error}: Failed (See Appendix 2)
\end{itemize}

\subsection{Terminate}
\textbf{[Function]} \texttt{func Terminate() error} \\
\textbf{[Purpose]} Releases ZKFinger SDK resources. \\
\textbf{[Parameter Description]} None \\
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{nil}: Succeeded
    \item \texttt{error}: Failed (See Appendix 2)
\end{itemize}

\subsection{GetDeviceCount}
\textbf{[Function]} \texttt{func GetDeviceCount() (int, error)} \\
\textbf{[Purpose]} Returns the number of connected fingerprint devices. \\
\textbf{[Parameter Description]} None \\
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{int, nil}: Connected device count (>= 0)
    \item \texttt{0, error}: Failed to query device count (See Appendix 2)
\end{itemize}

\subsection{OpenDevice}
\textbf{[Function]} \texttt{func OpenDevice(index int) (*Device, error)} \\
\textbf{[Purpose]} Opens a fingerprint device by its zero-based index. \\
\textbf{[Parameter Description]} 
\begin{itemize}
    \item \texttt{index}: Zero-based device index integer
\end{itemize}
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{*Device, nil}: Opened device handle instance
    \item \texttt{nil, error}: Failed to open device (returns \texttt{ErrOpenDevice})
\end{itemize}

\subsection{DBInit}
\textbf{[Function]} \texttt{func DBInit() (*DB, error)} \\
\textbf{[Purpose]} Creates an algorithm cache/database instance. \\
\textbf{[Parameter Description]} None \\
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{*DB, nil}: Initialized algorithm database handle instance
    \item \texttt{nil, error}: Initialization failed
\end{itemize}

\subsection{Device.AcquireFingerprint}
\textbf{[Method]} \texttt{func (d *Device) AcquireFingerprint(image []byte) ([]byte, error)} \\
\textbf{[Purpose]} Captures a fingerprint image and extracts its template. \\

\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{[]byte, nil}: Extracted fingerprint template slice trimmed to actual size
    \item \texttt{nil, error}: Capture or extraction failed (e.g., \texttt{ErrCaptureFailed})
\end{itemize}

\end{itemize}
\textbf{[Return Value]} 
\begin{itemize}
    \item \texttt{nil}: Succeeded
    \item \texttt{error}: Image capture failed (See Appendix 2)
\end{itemize}

\section{Usage Example}
\begin{lstlisting}
package main

import (
    "fmt"
    "log"
    
    "github.com/elmyrockers/go-zkfp"
)

func main() {
    // 1. Initialize the SDK
    if err := zkfp.Init(); err != nil {
        log.Fatalf("Init failed: %v", err)
    }
    defer zkfp.Terminate()

    // 2. Initialize the algorithm database/cache for matching & storage
    db, err := zkfp.DBInit()
    if err != nil {
        log.Fatalf("DBInit failed: %v", err)
    }
    defer db.Free()

    // 3. Check for connected devices
    count, err := zkfp.GetDeviceCount()
    if err != nil || count == 0 {
        log.Fatalf("No device found: %v", err)
    }

    // 4. Open the first device
    dev, err := zkfp.OpenDevice(0)
    if err != nil {
        log.Fatalf("OpenDevice failed: %v", err)
    }
    defer dev.Close()

    fmt.Println("Device opened and Algorithm DB initialized successfully!")

    // 5. Query required image buffer size and capture fingerprint
    imgSizeBytes, err := dev.ImageSize()
    if err != nil {
        log.Fatalf("Failed to get image size: %v", err)
    }

    imgBuf := make([]byte, imgSizeBytes)
    fmt.Println("Please place your finger on the scanner...")

    template, err := dev.AcquireFingerprint(imgBuf)
    if err != nil {
        log.Fatalf("Fingerprint capture failed: %v", err)
    }

    // 6. Add the captured template to the algorithm database with ID 1
    var fingerprintID uint = 1
    if err := db.Add(fingerprintID, template); err != nil {
        log.Fatalf("Failed to add template to DB: %v", err)
    }

    totalCount, _ := db.Count()
    fmt.Printf("Fingerprint successfully registered! Total records in DB: %d\n", totalCount)
}
\end{lstlisting}
